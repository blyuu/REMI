cmake_minimum_required(VERSION 3.25)
project(REMI_SceneComparison LANGUAGES CXX)
if(NOT MSVC OR NOT REMI_ROOT OR NOT REMI_REFERENCE_DIR)
    message(FATAL_ERROR "Use compare_scene.ps1 with MSVC and a reference source directory")
endif()
foreach(variant IN ITEMS baseline current)
    add_executable(scene_${variant} main.cpp "${REMI_ROOT}/engine/src/core/Math.cpp")
    if(variant STREQUAL "baseline")
        target_sources(scene_${variant} PRIVATE "${REMI_REFERENCE_DIR}/Scene.cpp")
        target_include_directories(scene_${variant} PRIVATE "${REMI_REFERENCE_DIR}/include")
    else()
        target_sources(scene_${variant} PRIVATE "${REMI_ROOT}/engine/src/scene/Scene.cpp")
    endif()
    target_include_directories(scene_${variant} PRIVATE "${REMI_ROOT}/engine/include")
    target_compile_features(scene_${variant} PRIVATE cxx_std_20)
    target_compile_options(scene_${variant} PRIVATE /W4 /WX /permissive- /utf-8)
    target_compile_definitions(scene_${variant} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN BENCHMARK_VARIANT="${variant}")
endforeach()
