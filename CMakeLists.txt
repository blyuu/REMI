cmake_minimum_required(VERSION 3.25)
project(REMI VERSION 0.11.0 LANGUAGES CXX)

if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "REMI currently supports Windows with MSVC. Use the vs2022-x64 preset.")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "REMI requires a 64-bit toolchain.")
endif()

include(CTest)
set(CMAKE_CXX_EXTENSIONS OFF)
option(REMI_STATIC_RUNTIME "Link the MSVC runtime statically for portable distribution" OFF)
if(REMI_STATIC_RUNTIME)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
endif()
option(REMI_ENABLE_ASAN "Enable MSVC AddressSanitizer for CPU lifetime regressions" OFF)
if(REMI_ENABLE_ASAN)
    string(REPLACE "/RTC1" "" CMAKE_CXX_FLAGS_DEBUG "${CMAKE_CXX_FLAGS_DEBUG}")
endif()

add_library(remi_build_options INTERFACE)
target_compile_features(remi_build_options INTERFACE cxx_std_20)
target_compile_options(remi_build_options INTERFACE /W4 /WX /permissive- /utf-8 /Zc:__cplusplus)
target_compile_definitions(remi_build_options INTERFACE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
if(REMI_ENABLE_ASAN)
    target_compile_options(remi_build_options INTERFACE /fsanitize=address /Zi)
    target_link_options(remi_build_options INTERFACE /INCREMENTAL:NO)
    target_compile_definitions(remi_build_options INTERFACE REMI_ASAN=1)
endif()

add_subdirectory(engine)
add_subdirectory(apps/bootstrap)
add_subdirectory(apps/sandbox)
add_subdirectory(games/gravity)
add_subdirectory(games/relay)
add_subdirectory(games/village)
add_subdirectory(tools/benchmarks)
set_property(DIRECTORY PROPERTY VS_STARTUP_PROJECT REMIGravity)

if(BUILD_TESTING)
    add_test(NAME Phase0.Bootstrap COMMAND REMIBootstrap)
    set_tests_properties(Phase0.Bootstrap PROPERTIES
        PASS_REGULAR_EXPRESSION "Phase 0 bootstrap OK")
    add_subdirectory(tests)
endif()

install(TARGETS REMIGravity REMIRelay REMISandbox RUNTIME DESTINATION .)
install(FILES shaders/Basic.hlsl DESTINATION shaders)
install(FILES games/gravity/gravity.project.json DESTINATION .)
install(FILES assets/fonts/Pretendard-SemiBold.otf assets/fonts/LICENSE.txt
    assets/fonts/NotoSansKR-VF.ttf assets/fonts/NotoSansKR-OFL.txt DESTINATION assets/fonts)
install(FILES tests/fixtures/static_triangle.glb tests/fixtures/sample.png DESTINATION samples)
install(DIRECTORY tools/package/runtime/ DESTINATION .)
